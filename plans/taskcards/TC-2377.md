## Objective

W9 (Validator) writes `quality_feedback.json` after each run. W4 reads it on the next run to
increase top_k claim selection for pages that had insufficient content. W2 reads it to lower
snippet similarity threshold for pages with low snippet coverage.

**Feature flag**: `use_feedback: false` in run_config (default false — zero impact on existing tests).

**Pre-condition**: Run both pilots 3× consecutively with TC-2376, TC-2378, TC-2379 complete before
starting this TC. Requires stable feedback data as input.

## Governance (spec first)

Create `specs/42_quality_feedback_loop.md` defining:

- `quality_feedback.json` schema:
  ```json
  {
    "run_id": "string",
    "pages": [
      {
        "output_path": "string",
        "gate_issues": [{"error_code": "string", "severity": "string"}],
        "failed_checks": ["string"],
        "suggested_actions": ["increase_claim_count", "low_snippet_coverage"]
      }
    ]
  }
  ```
- W9 emitter: write after validation pass (feature-flag guarded)
- W4 consumer: if `increase_claim_count` for a page → increase top_k by +3 (max 20)
- W2 consumer: if `low_snippet_coverage` → lower snippet similarity threshold by 0.05 (min 0.2)
- Immutability: feedback from run N is read-only in run N+1 (never overwrite mid-run)
- Feature flag: `use_feedback: false` in run_config

## W9 Emitter

In `w9_validator/worker.py`, after all gates run:

```python
def emit_quality_feedback(
    run_dir: Path,
    run_id: str,
    page_results: List[Dict],
    use_feedback: bool = False,
) -> None:
    """Write quality_feedback.json for next-run consumption."""
    if not use_feedback:
        return

    def _suggest_actions(result: Dict) -> List[str]:
        actions = []
        issues = result.get("issues", [])
        failed = result.get("failed_checks", [])
        if "content_density" in failed or any(i.get("error_code", "").startswith("G7") for i in issues):
            actions.append("increase_claim_count")
        if "snippet_attribution" in failed or any(i.get("error_code", "").startswith("G3") for i in issues):
            actions.append("low_snippet_coverage")
        return actions

    feedback = {
        "run_id": run_id,
        "pages": [
            {
                "output_path": r.get("output_path", ""),
                "gate_issues": [
                    i for i in r.get("issues", [])
                    if i.get("severity") in ("error", "critical")
                ],
                "failed_checks": r.get("failed_checks", []),
                "suggested_actions": _suggest_actions(r),
            }
            for r in page_results
        ],
    }

    feedback_path = run_dir / "work" / "quality_feedback.json"
    feedback_path.write_text(json.dumps(feedback, indent=2), encoding="utf-8")
    logger.info("quality_feedback_emitted", path=str(feedback_path), pages=len(feedback["pages"]))
```

## W4 Consumer

In `w4_ia_planner/worker.py`:

```python
def _load_quality_feedback(run_dir: Path) -> Dict:
    """Load quality_feedback.json from previous run."""
    feedback_path = run_dir / "work" / "quality_feedback.json"
    if not feedback_path.exists():
        return {}
    try:
        return json.loads(feedback_path.read_text(encoding="utf-8"))
    except Exception:
        return {}

def _adjust_top_k_from_feedback(
    page_slug: str,
    default_top_k: int,
    feedback: Dict,
) -> int:
    """Increase top_k if previous run indicated insufficient claims."""
    for page in feedback.get("pages", []):
        if page["output_path"].endswith(page_slug):
            if "increase_claim_count" in page.get("suggested_actions", []):
                return min(default_top_k + 3, 20)
    return default_top_k
```

## W2 Consumer

In `w2_facts_builder/worker.py`:

```python
def _adjust_snippet_threshold_from_feedback(
    default_threshold: float,
    feedback: Dict,
) -> float:
    """Lower snippet similarity threshold if previous run had low snippet coverage."""
    pages_with_low_snippets = [
        p for p in feedback.get("pages", [])
        if "low_snippet_coverage" in p.get("suggested_actions", [])
    ]
    if len(pages_with_low_snippets) > 2:  # Only adjust if multiple pages affected
        adjusted = max(default_threshold - 0.05, 0.2)
        logger.info("snippet_threshold_adjusted from=%.2f to=%.2f", default_threshold, adjusted)
        return adjusted
    return default_threshold
```
