## Objective

Add a Layer 1 LLM response validator that catches defects at call time (not in Gate 16) and retries
with error context before bad responses propagate downstream.

Reference implementation: `C:/Users/prora/OneDrive/Documents/GitHub/content-generator/src/utils/llm_response_validator.py`
and `src/services/services.py:405`.

## Background

Currently, fence imbalance and frontmatter contamination are detected only at Gate 16 (W9), long after
the response has passed through W5, W6, W7, and W8. The content-generator validates immediately after
each LLM call and retries with an enhanced prompt explaining the error. This catches ~70% of format
errors before they reach the sanitizer pipeline.

## Implementation

### New file: `src/launch/workers/_shared/llm_response_validator.py`

```python
from dataclasses import dataclass, field
from typing import List
import time

@dataclass
class ValidationResult:
    is_valid: bool
    errors: List[str] = field(default_factory=list)
    warnings: List[str] = field(default_factory=list)
    content_type: str = "unknown"
    validation_duration_ms: int = 0

def validate_llm_response(content: str, content_type: str = "markdown") -> ValidationResult:
    """Layer 1: fast validation at LLM call time. Target: <15ms."""
    t0 = time.monotonic()
    errors, warnings = [], []

    # 1. Code fence balance (content-generator llm_response_validator.py:55)
    fence_count = content.count("```")
    if fence_count % 2 != 0:
        errors.append(f"Unbalanced code fences: {fence_count} markers (must be even)")

    # 2. Frontmatter contamination (duplicate --- blocks)
    frontmatter_blocks = [i for i, ln in enumerate(content.split("\n")) if ln.strip() == "---"]
    if len(frontmatter_blocks) > 2:
        errors.append(f"Frontmatter contamination: {len(frontmatter_blocks)} --- blocks found")

    # 3. Truncation detection
    stripped = content.rstrip()
    if stripped and stripped[-1] not in ".!?`>":
        warnings.append("Response may be truncated (does not end with punctuation or fence)")

    # 4. Minimum content
    if len(content.strip()) < 50:
        errors.append(f"Response too short: {len(content.strip())} chars (minimum 50)")

    return ValidationResult(
        is_valid=len(errors) == 0,
        errors=errors,
        warnings=warnings,
        content_type=content_type,
        validation_duration_ms=int((time.monotonic() - t0) * 1000)
    )

def enhance_prompt_for_retry(base_prompt: str, validation_result: ValidationResult) -> str:
    """Add error context to prompt on retry (content-generator services.py:470 pattern)."""
    if not validation_result.errors:
        return base_prompt
    error_block = "\n".join(f"- {e}" for e in validation_result.errors)
    return (
        f"{base_prompt}\n\n"
        f"IMPORTANT: Previous attempt had these issues — fix them:\n{error_block}\n"
        f"Ensure code fences are balanced (open and close with ```).\n"
        f"Do not wrap the entire response in a code fence.\n"
        f"Complete all sentences before the response ends."
    )
```

Find where the raw LLM response is returned in `src/launch/services/llm_provider.py` (or equivalent

```python
from src.launch.workers._shared.llm_response_validator import validate_llm_response, enhance_prompt_for_retry

MAX_L1_RETRIES = 2  # At most 2 retries before returning best effort

for attempt in range(MAX_L1_RETRIES + 1):
    raw = _call_llm_raw(prompt, **kwargs)
    result = validate_llm_response(raw, content_type=kwargs.get("content_type", "markdown"))
    if result.is_valid:
        return raw
    if attempt < MAX_L1_RETRIES:
        logger.warning("L1_VALIDATOR_FAIL attempt=%d errors=%s", attempt, result.errors)
        prompt = enhance_prompt_for_retry(prompt, result)
        continue
    logger.error("L1_VALIDATOR_FAIL_FINAL errors=%s", result.errors)
    return raw  # Return best effort; downstream gates catch remaining issues
```
