## Objective

W1 tracks file hashes between runs. Files whose content hash hasn't changed are skipped during
re-ingestion, reducing W1/W2 processing time by ~70% for large repos on subsequent runs.

```python
"""Ingestion state manager — tracks file hashes for incremental processing.

Reference: content-generator kb_ingestion.py IngestionStateManager
"""
from __future__ import annotations
import hashlib
import json
import logging
from pathlib import Path
from typing import Optional

logger = logging.getLogger(__name__)


class IngestionStateManager:
    """Tracks per-file SHA256 hashes to skip unchanged files on re-ingestion.

    State persists between runs via a JSON file. Thread-safe for single-process use.
    """

    def __init__(self, state_file: Path) -> None:
        self.state_file = state_file
        self._state: dict = {}
        if state_file.exists():
            try:
                self._state = json.loads(state_file.read_text(encoding="utf-8"))
            except (json.JSONDecodeError, OSError) as e:
                logger.warning("ingestion_state_load_fail path=%s error=%s", state_file, e)
                self._state = {}

    def needs_ingestion(self, file_path: Path) -> bool:
        """Return True if file is new or content has changed since last ingestion."""
        key = str(file_path)
        stored_hash = self._state.get(key)
        if stored_hash is None:
            return True
        try:
            current_hash = self._file_hash(file_path)
            return stored_hash != current_hash
        except OSError:
            return True  # File inaccessible → re-ingest to be safe

    def mark_ingested(self, file_path: Path) -> None:
        """Record file hash after successful ingestion. Writes state file."""
        key = str(file_path)
        try:
            self._state[key] = self._file_hash(file_path)
            self._persist()
        except OSError as e:
            logger.warning("ingestion_state_mark_fail path=%s error=%s", file_path, e)

    def clear(self) -> None:
        """Force re-ingestion of all files on next run."""
        self._state = {}
        self._persist()

    @property
    def tracked_file_count(self) -> int:
        return len(self._state)

    def _file_hash(self, path: Path) -> str:
        return hashlib.sha256(path.read_bytes()).hexdigest()[:16]

    def _persist(self) -> None:
        try:
            self.state_file.write_text(
                json.dumps(self._state, indent=2),
                encoding="utf-8",
            )
        except OSError as e:
            logger.warning("ingestion_state_persist_fail error=%s", e)
```

## W1 worker.py Integration

Read `worker.py` first. In the file-scanning loop, add:

```python
from launch.workers.w1_repo_scout.ingestion_state import IngestionStateManager

# In run() or equivalent, before the file loop:
state_file = run_dir / "work" / "ingestion_state.json"
ingestion_state = IngestionStateManager(state_file)
skipped = 0
ingested = 0

# In the file loop:
for file_path in candidate_files:
    if not ingestion_state.needs_ingestion(file_path):
        skipped += 1
        continue
    # ... existing ingestion logic ...
    ingestion_state.mark_ingested(file_path)
    ingested += 1

logger.info("w1_incremental_ingestion", ingested=ingested, skipped=skipped,
            total=ingested + skipped)
```

**Note**: State file is at `{run_dir}/work/ingestion_state.json`. This persists between runs
that share the same `run_dir`. For fresh runs, state is empty → all files re-ingested.
