Create this file with:

```python
"""KB source chunking for retrieval-augmented generation (D-3).

Chunks prose documentation files at semantic boundaries for W5 grounding.
Only processes prose formats (.md, .rst, .txt, .yaml, .yml) —
code files are handled by W3 SnippetCurator.

Reference: content-generator src/engine/engine.py:567
"""
from __future__ import annotations
import hashlib
import re
from pathlib import Path
from typing import Any, Dict, Iterator, List

MIN_CHUNK_TOKENS = 30
MAX_CHUNK_TOKENS = 1000
OVERLAP_TOKENS = 200
MAX_FILE_SIZE_BYTES = 5 * 1024 * 1024  # 5MB cap

CHUNK_EXTENSIONS = {".md", ".rst", ".txt", ".yaml", ".yml"}
SKIP_PATTERNS = [
    r"test_", r"_test\.", r"node_modules", r"\.venv",
    r"__pycache__", r"\.git", r"\.pytest_cache",
]


def chunk_source_files(repo_dir: Path, max_chunks: int = 2000) -> List[Dict[str, Any]]:
    """Chunk prose documentation files at semantic boundaries."""
    chunks: List[Dict[str, Any]] = []
    for file_path in _iter_source_files(repo_dir):
        try:
            file_chunks = _chunk_file(file_path, repo_dir)
            chunks.extend(file_chunks)
            if len(chunks) >= max_chunks:
                return chunks[:max_chunks]
        except Exception:
            continue
    return chunks


def retrieve_relevant_chunks(query: str, chunks: List[Dict], top_k: int = 5) -> List[Dict]:
    """Retrieve top-K chunks relevant to a query using TF-IDF cosine similarity."""
    if not chunks:
        return []
    try:
        from launch.workers._shared.embeddings import tfidf_cosine_similarity
        texts = [c["text"] for c in chunks]
        scores = tfidf_cosine_similarity(query, texts)
        ranked = sorted(zip(scores, chunks), key=lambda x: -x[0])
        return [c for _, c in ranked[:top_k]]
    except Exception:
        return chunks[:top_k]


def _iter_source_files(repo_dir: Path) -> Iterator[Path]:
    for path in sorted(repo_dir.rglob("*")):
        if not path.is_file():
            continue
        if path.suffix not in CHUNK_EXTENSIONS:
            continue
        rel = str(path.relative_to(repo_dir))
        if any(re.search(p, rel) for p in SKIP_PATTERNS):
            continue
        try:
            if path.stat().st_size > MAX_FILE_SIZE_BYTES:
                continue
        except OSError:
            continue
        yield path


def _chunk_file(file_path: Path, repo_dir: Path) -> List[Dict[str, Any]]:
    try:
        text = file_path.read_text(encoding="utf-8", errors="ignore")
    except OSError:
        return []
    if not text.strip():
        return []
    rel_path = str(file_path.relative_to(repo_dir))
    if file_path.suffix in {".yaml", ".yml"}:
        return _chunk_yaml(text, rel_path)
    return _chunk_by_headings(text, rel_path)


def _chunk_by_headings(text: str, rel_path: str) -> List[Dict[str, Any]]:
    """Split at H1-H3 heading boundaries; sub-chunk long sections at paragraphs."""
    sections = re.split(r'(?m)(?=^#{1,3} )', text)
    chunks = []
    for section in sections:
        section = section.strip()
        if not section:
            continue
        m = re.match(r'^(#{1,3}) (.+)', section)
        heading_context = m.group(2).strip() if m else ""
        token_count = len(section.split())
        if token_count < MIN_CHUNK_TOKENS:
            continue
        if token_count <= MAX_CHUNK_TOKENS:
            chunks.append(_make_chunk(section, rel_path, heading_context))
        else:
            chunks.extend(_sub_chunk_paragraphs(section, rel_path, heading_context))
    return chunks


def _sub_chunk_paragraphs(
    section_text: str, rel_path: str, heading_context: str
) -> List[Dict[str, Any]]:
    """Split long sections at paragraph boundaries with overlap."""
    paragraphs = re.split(r'\n\n+', section_text)
    chunks = []
    current_parts: List[str] = []
    current_tokens = 0
    overlap_buffer: List[str] = []

    for para in paragraphs:
        para_tokens = len(para.split())
        if current_tokens + para_tokens > MAX_CHUNK_TOKENS and current_parts:
            chunk_text = "\n\n".join(current_parts)
            if len(chunk_text.split()) >= MIN_CHUNK_TOKENS:
                chunks.append(_make_chunk(chunk_text, rel_path, heading_context))
            # Keep last OVERLAP_TOKENS worth of content for next chunk
            overlap_words = " ".join(chunk_text.split()[-OVERLAP_TOKENS:])
            current_parts = [overlap_words, para] if overlap_words else [para]
            current_tokens = len(overlap_words.split()) + para_tokens
        else:
            current_parts.append(para)
            current_tokens += para_tokens

    if current_parts:
        chunk_text = "\n\n".join(current_parts)
        if len(chunk_text.split()) >= MIN_CHUNK_TOKENS:
            chunks.append(_make_chunk(chunk_text, rel_path, heading_context))
    return chunks


def _chunk_yaml(text: str, rel_path: str, lines_per_chunk: int = 50) -> List[Dict[str, Any]]:
    lines = text.splitlines()
    chunks = []
    for i in range(0, len(lines), lines_per_chunk):
        block = "\n".join(lines[i:i + lines_per_chunk]).strip()
        if len(block.split()) < MIN_CHUNK_TOKENS:
            continue
        m = re.search(r'^([a-zA-Z_][a-zA-Z0-9_]*):', block, re.MULTILINE)
        heading_context = m.group(1) if m else ""
        chunks.append(_make_chunk(block, rel_path, heading_context))
    return chunks


def _make_chunk(text: str, rel_path: str, heading_context: str) -> Dict[str, Any]:
    chunk_id = hashlib.sha256(
        f"{rel_path}:{heading_context}:{text[:60]}".encode()
    ).hexdigest()[:16]
    return {
        "chunk_id": chunk_id,
        "file_path": rel_path,
        "text": text,
        "heading_context": heading_context,
        "token_count": len(text.split()),
    }
```


Add method:
```python
def _load_source_chunks(self, run_dir):
    if self._source_chunks_loaded or run_dir is None:
        return
    self._source_chunks_loaded = True
    try:
        source_chunks_path = Path(run_dir) / "artifacts" / "source_chunks.json"
        if source_chunks_path.exists():
            import json
            with open(source_chunks_path, "r", encoding="utf-8") as f:
                data = json.load(f)
            self._source_chunks = data.get("chunks", [])
    except Exception as e:
        logger.warning(f"Could not load source chunks: {e}")
```
